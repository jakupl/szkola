Przykładowy algorytm KNN
bez głosowania:
```
punkty = [
    (1, 2, "A"),
    (5, 7, "A"),
    (2, 5, "B"), 
    (8, 2, "B"), 
    (5, 7, "C")]


def knn(nowy_X, nowy_Y):
    naj_etykieta = None
    naj_dist = None

    for x, y, etykieta in punkty:
        d = (x - nowy_X) ** 2 + (y - nowy_Y) ** 2

        if naj_dist is None or d < naj_dist:
            naj_dist = d
            naj_etykieta = etykieta

    print(f"pkt nalezy do: {naj_etykieta}")


knn(4, 2)
```
z głosowaniem:
```
def knn(k, zbior, punkt):
    odl = lambda a: sum((x - y) ** 2 for x, y in zip(a, punkt))
    glosy = [zbior[a] for a in sorted(zbior, key=odl)[:k]]
    return max(glosy, key=glosy.count)

zbior = {(2, 2, 3): "b", (2, 3, 2): "b", (1, 1, 1): "a", (9, 9, 9): "c", (5, 5, 5): "a"}
print(knn(3, zbior, (2, 2, 2)))  # b
```
